    84	def build_folds(start: date = DEFAULT_START, n_folds: int = N_FOLDS) -> list[Fold]:
    85	    """Build the n_folds 3-way walk-forward calendar.
    86	
    87	    Fold i (1-indexed), s_i = start + (i-1)*STEP_DAYS:
    88	        train = [s_i,            s_i+42d)
    89	        val   = [s_i+42d,        s_i+56d)
    90	        test  = [s_i+56d,        s_i+70d)
    91	    Half-open intervals (end exclusive).
    92	    """
    93	    folds: list[Fold] = []
    94	    for i in range(n_folds):
    95	        s = start + timedelta(days=i * STEP_DAYS)
    96	        train_start = s
    97	        train_end_excl = s + timedelta(days=TRAIN_DAYS)
    98	        val_start = train_end_excl
    99	        val_end_excl = val_start + timedelta(days=VAL_DAYS)
   100	        test_start = val_end_excl
   101	        test_end_excl = test_start + timedelta(days=TEST_DAYS)
   102	        folds.append(
   103	            Fold(
   104	                fold_id=i + 1,
   105	                train_start=train_start,
   106	                train_end_excl=train_end_excl,
   107	                val_start=val_start,
   108	                val_end_excl=val_end_excl,
   109	                test_start=test_start,
   110	                test_end_excl=test_end_excl,
   111	                pretest_start=train_start,
   112	                pretest_end_excl=test_start,
   113	            )
   114	        )
   115	    return folds

   120	def classify_trend(now: Optional[float], past: Optional[float]) -> str:
   121	    """30d trend bucket. BULL >+10%, BEAR <-10%, else NEUTRAL. Bad data -> UNKNOWN."""
   122	    for v in (now, past):
   123	        if v is None or isinstance(v, bool) or not isinstance(v, (int, float)) or not np.isfinite(v) or v <= 0:
   124	            return "UNKNOWN"
   125	    pct = (now - past) / past
   126	    if pct > 0.10:
   127	        return "BULL"
   128	    if pct < -0.10:
   129	        return "BEAR"
   130	    return "NEUTRAL"

   133	def classify_vol(now: Optional[float], train_series: list[float]) -> str:
   134	    """DVOL bucket vs train-window p33/p66. Bad/insufficient data -> UNKNOWN."""
   135	    if now is None or isinstance(now, bool) or not isinstance(now, (int, float)) or not np.isfinite(now) or now <= 0:
   136	        return "UNKNOWN"
   137	    arr = np.array([v for v in (train_series or []) if isinstance(v, (int, float)) and not isinstance(v, bool) and np.isfinite(v) and v > 0])
   138	    if len(arr) < 5:
   139	        return "UNKNOWN"
   140	    p33, p66 = np.quantile(arr, [0.33, 0.66])
   141	    if now > p66:
   142	        return "HIGH"
   143	    if now < p33:
   144	        return "LOW"
   145	    return "MID"

   148	def tag_fold_regime(fold: Fold, market_data_fn: Optional[Callable[[date], dict]]) -> dict:
   149	    """Regime tags at test_start. market_data_fn(date)->{btc_price,hl_price,btc_dvol} or None.
   150	
   151	    30d trend lookback uses [test_start-30d, test_start) (NOT train_start; V13 m09 r1 bug).
   152	    DVOL percentiles computed over the train window [train_start, train_end_excl).
   153	    No market_data_fn -> all UNKNOWN (regime is reporting-only; absence never blocks M3).
   154	    """
   155	    if market_data_fn is None:
   156	        return {"btc_trend_bucket": "UNKNOWN", "hl_trend_bucket": "UNKNOWN", "dvol_bucket": "UNKNOWN"}
   157	
   158	    def _get(d: date, key: str):
   159	        md = market_data_fn(d)
   160	        if not md:
   161	            return None
   162	        v = md.get(key)
   163	        if v is None or isinstance(v, bool) or not isinstance(v, (int, float)) or not np.isfinite(v) or v <= 0:
   164	            return None
   165	        return float(v)
   166	
   167	    test_start = fold.test_start
   168	    lb = test_start - timedelta(days=30)
   169	    btc_trend = classify_trend(_get(test_start, "btc_price"), _get(lb, "btc_price"))
   170	    hl_trend = classify_trend(_get(test_start, "hl_price"), _get(lb, "hl_price"))
   171	
   172	    dvol_series = []
   173	    d = fold.train_start
   174	    while d < fold.train_end_excl:
   175	        v = _get(d, "btc_dvol")
   176	        if v is not None:
   177	            dvol_series.append(v)
   178	        d += timedelta(days=1)
   179	    dvol_bucket = classify_vol(_get(test_start, "btc_dvol"), dvol_series)
   180	    return {"btc_trend_bucket": btc_trend, "hl_trend_bucket": hl_trend, "dvol_bucket": dvol_bucket}

   212	def build_activity(
   213	    folds: list[Fold],
   214	    actions: pd.DataFrame,
   215	    journeys: pd.DataFrame,
   216	    key_col: str = "wallet",
   217	    journeys_complete: bool = True,
   218	) -> tuple[pd.DataFrame, pd.DataFrame]:
   219	    """Per (key, fold, window) action/journey counts + active flags, and a per-key summary.
   220	
   221	    activity_basis is GLOBAL: 'journey' when journeys input present+complete, else
   222	    'action_fallback' (codex r1 fix #4 — NEVER per-key on zero rows; zero rows = inactive).
   223	    `active` = >=1 journey OPENED (entry_ts) in the window (or action fallback). Open/unclosed
   224	    journeys count (they carry an entry_ts).
   225	    """
   226	    # codex code-r1 #4: complete journeys input present => GLOBAL 'journey' even with zero rows
   227	    # (zero rows = genuinely inactive, not missing data). Fallback only when input absent/incomplete.
   228	    basis = "journey" if (journeys_complete and journeys is not None) else "action_fallback"
   229	
   230	    def _to_dt(s):
   231	        # codex code-r1 #5: pd.api.types, not np.issubdtype (handles extension/tz/nullable dtypes).
   232	        return pd.to_datetime(s, unit="ms", utc=True) if pd.api.types.is_numeric_dtype(s) else pd.to_datetime(s, utc=True)
   233	
   234	    a = actions.copy()
   235	    if "stream_replay_valid" in a.columns:
   236	        a = a[a["stream_replay_valid"].fillna(False).astype(bool)].copy()
   237	    a["ts"] = _to_dt(a["ts"])
   238	    use_journeys = basis == "journey"
   239	    if use_journeys:
   240	        j = journeys.copy()
   241	        if "lifecycle_valid" in j.columns:
   242	            j = j[j["lifecycle_valid"].fillna(False).astype(bool)].copy()
   243	        if "stream_replay_valid" in j.columns:
   244	            j = j[j["stream_replay_valid"].fillna(False).astype(bool)].copy()
   245	        # journey OPEN timestamp = entry_ts (ms). Open/unclosed journeys still have entry_ts.
   246	        j["open_ts"] = _to_dt(j["entry_ts"])
   247	    else:
   248	        j = None
   249	
   250	    keys = pd.Index(a[key_col].unique())
   251	    if use_journeys:
   252	        keys = keys.union(pd.Index(j[key_col].unique()))
   253	
   254	    fold_rows = []
   255	    for f in folds:
   256	        for w in WINDOWS:
   257	            s, e = f.window_bounds(w)
   258	            s_ts, e_ts = pd.Timestamp(s, tz="UTC"), pd.Timestamp(e, tz="UTC")
   259	            am = a[(a["ts"] >= s_ts) & (a["ts"] < e_ts)]
   260	            n_actions = am.groupby(key_col).size()
   261	            if use_journeys:
   262	                jm = j[(j["open_ts"] >= s_ts) & (j["open_ts"] < e_ts)]
   263	                n_journeys = jm.groupby(key_col).size()
   264	                first_ts = am.groupby(key_col)["ts"].min()
   265	                last_ts = am.groupby(key_col)["ts"].max()
   266	            else:
   267	                n_journeys = pd.Series(dtype=int)
   268	                first_ts = am.groupby(key_col)["ts"].min()
   269	                last_ts = am.groupby(key_col)["ts"].max()
   270	            df = pd.DataFrame({"key": keys})
   271	            df["fold_id"] = f.fold_id
   272	            df["window"] = w
   273	            df["n_actions"] = df["key"].map(n_actions).fillna(0).astype(int)
   274	            df["n_journeys"] = df["key"].map(n_journeys).fillna(0).astype(int)
   275	            if use_journeys:
   276	                df["active"] = df["n_journeys"] >= 1
   277	            else:
   278	                df["active"] = df["n_actions"] >= 1
   279	            # dict mapper (NOT a datetime-valued Series mapper — that hits a pandas
   280	            # _map_values cast bug). pd.to_datetime forces datetime so the pivot does not
   281	            # collide float64 (empty windows) with datetime64 (populated windows).
   282	            df["first_action_ts"] = pd.to_datetime(df["key"].map(first_ts.to_dict()), utc=True)
   283	            df["last_action_ts"] = pd.to_datetime(df["key"].map(last_ts.to_dict()), utc=True)
   284	            fold_rows.append(df)
   285	    long = pd.concat(fold_rows, ignore_index=True)
   286	
   287	    # Raw per-key totals over the WHOLE archive (codex code-r1 #9: NOT summed across folds —
   288	    # overlapping train/val windows would double-count the same action/journey).
   289	    total_actions_raw = a.groupby(key_col).size()
   290	    total_journeys_raw = j.groupby(key_col).size() if use_journeys else pd.Series(dtype=int)
   291	    first_seen_raw = a.groupby(key_col)["ts"].min()
   292	    last_seen_raw = a.groupby(key_col)["ts"].max()
   293	
   294	    # Pivot to wide per (key, fold).
   295	    wide = _pivot_wide(long, keys, folds, basis)
   296	    summary = _summarize(wide, keys, basis, total_actions_raw, total_journeys_raw, first_seen_raw, last_seen_raw)
   297	    return wide, summary

   299	
   300	def _pivot_wide(long: pd.DataFrame, keys: pd.Index, folds: list[Fold], basis: str) -> pd.DataFrame:
   301	    out = []
   302	    for fid in [f.fold_id for f in folds]:
   303	        block = long[long["fold_id"] == fid]
   304	        piv = block.pivot(index="key", columns="window")
   305	        row = pd.DataFrame(index=keys)
   306	        row.index.name = "key"
   307	        for w in WINDOWS:
   308	            row[f"n_actions_{w}"] = piv[("n_actions", w)] if ("n_actions", w) in piv else 0
   309	            row[f"n_journeys_{w}"] = piv[("n_journeys", w)] if ("n_journeys", w) in piv else 0
   310	            row[f"active_{w}"] = piv[("active", w)] if ("active", w) in piv else False
   311	        for w in ("train", "val", "test"):
   312	            row[f"first_action_ts_{w}"] = piv[("first_action_ts", w)] if ("first_action_ts", w) in piv else pd.NaT
   313	            row[f"last_action_ts_{w}"] = piv[("last_action_ts", w)] if ("last_action_ts", w) in piv else pd.NaT
   314	        row["active_any"] = row[[f"active_{w}" for w in ("train", "val", "test")]].any(axis=1)
   315	        row = row.reset_index()
   316	        row.insert(1, "fold_id", fid)
   317	        out.append(row)
   318	    wide = pd.concat(out, ignore_index=True)
   319	    wide["key_kind"] = "wallet"
   320	    wide["activity_basis"] = basis
   321	    for w in WINDOWS + ("any",):
   322	        col = f"active_{w}"
   323	        if col in wide:
   324	            wide[col] = wide[col].fillna(False).astype(bool)
   325	    for w in WINDOWS:
   326	        wide[f"n_actions_{w}"] = wide[f"n_actions_{w}"].fillna(0).astype(int)
   327	        wide[f"n_journeys_{w}"] = wide[f"n_journeys_{w}"].fillna(0).astype(int)
   328	    # codex code-r1 #8: contract column order key, key_kind, fold_id, ...
   329	    lead = ["key", "key_kind", "fold_id"]
   330	    wide = wide[lead + [c for c in wide.columns if c not in lead]]
   331	    return wide
   332	
   333	
   334	def _summarize(
   335	    wide: pd.DataFrame,
   336	    keys: pd.Index,
   337	    basis: str,
   338	    total_actions_raw: pd.Series,
   339	    total_journeys_raw: pd.Series,
   340	    first_seen_raw: pd.Series,
   341	    last_seen_raw: pd.Series,
   342	) -> pd.DataFrame:
   343	    g = wide.groupby("key")
   344	    summary = pd.DataFrame({"key": list(keys)})
   345	    summary["key_kind"] = "wallet"
   346	    for w in WINDOWS:
   347	        summary[f"active_{w}_folds"] = summary["key"].map(g[f"active_{w}"].sum()).fillna(0).astype(int)
   348	    summary["active_any_folds"] = summary["key"].map(g["active_any"].sum()).fillna(0).astype(int)
   349	    # G5: active in >=3 TEST windows.
   350	    summary["active_folds_for_g5"] = summary["active_test_folds"]
   351	    # codex code-r1 #9: raw archive-wide totals (NOT fold-summed — overlapping train/val
   352	    # windows would double-count the same source row).
   353	    summary["total_actions"] = summary["key"].map(total_actions_raw.to_dict()).fillna(0).astype(int)
   354	    summary["total_journeys"] = summary["key"].map(total_journeys_raw.to_dict()).fillna(0).astype(int)
   355	    summary["first_seen_ts"] = pd.to_datetime(summary["key"].map(first_seen_raw.to_dict()), utc=True)
   356	    summary["last_seen_ts"] = pd.to_datetime(summary["key"].map(last_seen_raw.to_dict()), utc=True)
   357	    summary["activity_basis"] = basis
   358	    return summary

   101	    med_eq = float(raw_eq.median()) if len(raw_eq) else 0.0
   102	    if len(d) < 2:
   103	        return {"roe": 0.0, "max_dd": 0.0, "n_days": len(d), "structural_ruin": True,
   104	                "median_equity": med_eq, "frac_days_green": 0.0, "n_active_days": 0,
   105	                "median_leverage": 0.0, "n_incomplete_excluded": n_incomplete_excluded}
   106	    e = d["equity_usd"].astype(float).fillna(0.0)
   107	    flow = d["ext_flow_cum"].astype(float).ffill().fillna(0.0).diff().fillna(0.0)
   108	    e_prev = e.shift(1)
   109	    denom = e_prev.where(e_prev > RUIN_EQUITY_FLOOR, np.nan)   # need >$1 base to define a return
   110	    r = ((e - e_prev - flow) / denom).replace([np.inf, -np.inf], np.nan).dropna()
   111	    # LEVERAGE (Alberto spec): median daily gross notional / equity over days with a real (>$1) base.
   112	    if "gross_position_notional_usd" in d.columns:
   113	        pv = d["gross_position_notional_usd"].astype(float)
   114	        lev = (pv / e.where(e > RUIN_EQUITY_FLOOR, np.nan)).replace([np.inf, -np.inf], np.nan).dropna()
   115	        median_leverage = float(lev.median()) if len(lev) else 0.0
   116	    elif "position_value_usd" in d.columns:
   117	        # Compatibility for unit fixtures only. Production CLI rejects an M1
   118	        # artifact without the gross field below.
   119	        pv = d["position_value_usd"].astype(float).abs()
   120	        lev = (pv / e.where(e > RUIN_EQUITY_FLOOR, np.nan)).replace([np.inf, -np.inf], np.nan).dropna()
   121	        median_leverage = float(lev.median()) if len(lev) else 0.0
   122	    else:
   123	        median_leverage = 0.0
   124	    if len(r) == 0:
   125	        return {"roe": 0.0, "max_dd": 0.0, "n_days": len(d), "structural_ruin": True,
   126	                "median_equity": med_eq, "frac_days_green": 0.0, "n_active_days": 0,
   127	                "median_leverage": median_leverage, "n_incomplete_excluded": n_incomplete_excluded}
   128	    # CONSISTENCY (Alberto spec): share of ACTIVE days (a real flow-adjusted return defined) that are green.
   129	    # "active" excludes flat/no-base days (r is already dropna'd to days with a >$1 prior base).
   130	    active = r[r != 0.0]
   131	    n_active = int(len(active))
   132	    frac_green = float((active > 0).mean()) if n_active else 0.0
   133	    # codex code-r1 #3: prepend baseline 1.0 so a first-day drop is counted in maxDD.
   134	    cum = pd.concat([pd.Series([1.0]), (1 + r).cumprod()], ignore_index=True)
   135	    peak = cum.cummax()
   136	    max_dd = float(abs(((cum - peak) / peak).min()))
   137	    roe = float(cum.iloc[-1] - 1)
   138	    # terminal collapse: started with a real base but ended ~0
   139	    start_eq = float(e[e > RUIN_EQUITY_FLOOR].iloc[0]) if (e > RUIN_EQUITY_FLOOR).any() else 0.0
   140	    ruin = (start_eq <= RUIN_EQUITY_FLOOR) or (float(e.iloc[-1]) <= RUIN_EQUITY_FLOOR and start_eq > 100)
   141	    return {"roe": roe, "max_dd": max_dd, "n_days": len(d), "structural_ruin": bool(ruin),
   142	            "median_equity": med_eq, "frac_days_green": frac_green, "n_active_days": n_active,
   143	            "median_leverage": median_leverage, "n_incomplete_excluded": n_incomplete_excluded}

   146	def journey_metrics(jr: pd.DataFrame, lo_ms: int, hi_ms: int, accessible: set | None) -> dict:
   147	    """Pretest journey metrics. CLOSED-in-pretest journeys (entry & exit in [lo,hi)) for
   148	    PnL/hold/count; accessibility by notional over those journeys.
   149	
   150	    Boundary-straddling week-holder guard (codex finding a): a position OPENED in pretest but still
   151	    OPEN at test_start (exit missing OR exit >= hi_ms) is invisible to median_hold_s. Its CENSORED
   152	    hold = (hi_ms - entry_ts) -- using only info up to hi_ms (no future exit, no look-ahead) -- is
   153	    surfaced as censored_max_hold_s so a multi-day/week hold straddling the boundary is rejected."""
   154	    if jr is None or len(jr) == 0:
   155	        return _empty_jm()
   156	    if "lifecycle_valid" in jr.columns:
   157	        jr = jr[jr["lifecycle_valid"].fillna(False)].copy()
   158	    if "stream_replay_valid" in jr.columns:
   159	        jr = jr[jr["stream_replay_valid"].fillna(False)].copy()
   160	        if len(jr) == 0:
   161	            return _empty_jm()
   162	    # censored open-at-test_start holds: opened in pretest, not closed inside pretest (still open at
   163	    # hi_ms). Censored hold = hi_ms - entry_ts (info only up to hi_ms). Used ONLY for the upper hold
   164	    # (week-holder) gate; never enters PnL/count/median.
   165	    opened = jr[(jr["entry_ts"] >= lo_ms) & (jr["entry_ts"] < hi_ms)]
   166	    straddling = opened[opened["exit_ts"].isna() | (opened["exit_ts"] >= hi_ms)]
   167	    censored_max_hold_s = (float((hi_ms - straddling["entry_ts"].astype("int64")).max()) / 1000.0
   168	                           if len(straddling) else 0.0)
   169	    # codex code-r1 #4: closed-in-pretest = entry AND exit both inside [lo, hi).
   170	    j = jr[(jr["entry_ts"] >= lo_ms) & (jr["entry_ts"] < hi_ms)
   171	           & jr["exit_ts"].notna() & (jr["exit_ts"] >= lo_ms) & (jr["exit_ts"] < hi_ms)].copy()
   172	    if len(j) == 0:
   173	        jm = _empty_jm()
   174	        jm["censored_max_hold_s"] = censored_max_hold_s
   175	        return jm
   176	    net_pnl = float(j["net_realized_pnl"].sum())
   177	    n_j = int(len(j))
   178	    dur_s = (j["duration_h"].astype(float) * 3600.0)   # m02 emits duration_h (not duration_hours)
   179	    median_hold_s = float(dur_s.median())
   180	    share_below = float((dur_s < P95_COPY_LATENCY_S).mean())
   181	    # accessibility by notional (max_position_notional proxy for journey size)
   182	    # codex code-r1 #5: index-aligned fallback (j.index) so the in_acc boolean mask aligns.
   183	    if "max_position_notional" in j.columns:
   184	        notional = j["max_position_notional"].astype(float).abs()
   185	    else:
   186	        notional = pd.Series(1.0, index=j.index)
   187	    if accessible is None:
   188	        acc_notional = float("nan"); acc_count = float("nan")  # unknown
   189	    else:
   190	        in_acc = j["coin"].isin(accessible)
   191	        tot = notional.sum()
   192	        acc_notional = float(notional[in_acc].sum() / tot) if tot > 0 else 1.0
   193	        acc_count = float(in_acc.mean())
   194	    return {"net_pnl": net_pnl, "n_journeys": n_j, "median_hold_s": median_hold_s,
   195	            "share_below_latency": share_below, "accessible_frac_notional": acc_notional,
   196	            "accessible_frac_count": acc_count, "censored_max_hold_s": censored_max_hold_s}

   199	def _empty_jm() -> dict:
   200	    return {"net_pnl": 0.0, "n_journeys": 0, "median_hold_s": 0.0, "share_below_latency": 1.0,
   201	            "accessible_frac_notional": float("nan"), "accessible_frac_count": float("nan"),
   202	            "censored_max_hold_s": 0.0}

   205	# Alberto 2026-06-09: M5 was doing TWO jobs -- copyability filtering (legit) AND performance pre-judging
   206	# (net_pnl/roe/days_green), which starves M6b ranking (only ~20 already-profitable wallets reach it) and
   207	# makes the pool artificially negative-mean OOS. COPYABILITY_ONLY mode (env M5_COPYABILITY_ONLY=1) keeps the
   208	# copyability/risk gates but does NOT let the performance gates block eligibility (they're still RECORDED in
   209	# fail_reasons for audit). This widens the eligible pool ~175 -> ~2,383 entity-folds so M6b can RANK
   210	# performance instead of M5 pre-gating it. Default OFF -> byte-identical to the strict pipeline.
   211	import os as _os
   212	COPYABILITY_ONLY = _os.environ.get("M5_COPYABILITY_ONLY", "0") == "1"
   213	# performance fails (judged by ranking, not eligibility) vs risk/copyability fails (hard exclusions)
   214	_PERF_FAIL_PREFIXES = ("net_pnl<=0", "roe<=0", "days_green<")

   217	def apply_floors(
   218	    eqm: dict, jm: dict, *, equity_required: bool = True,
   219	) -> tuple[bool, list[str]]:
   220	    """Per-fold pretest floors. Returns (eligible, fail_reasons).
   221	
   222	    In COPYABILITY_ONLY mode, performance fails (net_pnl/roe/days_green) are recorded in fail_reasons but do
   223	    NOT block eligibility -- only risk/copyability fails do. structural_ruin STAYS a hard risk gate.
   224	    """
   225	    f = []
   226	    if jm["net_pnl"] <= 0:
   227	        f.append(f"net_pnl<=0 ({jm['net_pnl']:.1f})")
   228	    if equity_required:
   229	        if eqm["structural_ruin"]:
   230	            f.append("structural_ruin")
   231	        elif eqm["roe"] <= 0:
   232	            f.append(f"roe<=0 ({eqm['roe']:.2%})")
   233	        # absolute pretest equity floor: drop tiny degen accounts that merely survived (fold-pure)
   234	        me = eqm.get("median_equity", 0.0)
   235	        if me < MIN_EQUITY_USD:
   236	            f.append(f"equity_too_small (${me:.0f}<{MIN_EQUITY_USD:.0f})")
   237	        if eqm["max_dd"] > MAXDD_CAP:
   238	            f.append(f"max_dd>{MAXDD_CAP:.0%} ({eqm['max_dd']:.0%})")
   239	    if jm["n_journeys"] < MIN_JOURNEYS_PRETEST:
   240	        f.append(f"n_journeys<{MIN_JOURNEYS_PRETEST} ({jm['n_journeys']})")
   241	    if jm["median_hold_s"] <= HOLD_FLOOR_S:
   242	        f.append(f"median_hold<={HOLD_FLOOR_S:.0f}s ({jm['median_hold_s']:.0f}s)")
   243	    # upper hold gate: V15 copies fast directional; reject multi-day/week holders
   244	    if jm["median_hold_s"] > SWING_MAX_HOLD_S:
   245	        f.append(f"hold_too_long (>{SWING_MAX_HOLD_S:.0f}s, {jm['median_hold_s']:.0f}s)")
   246	    # boundary-straddling week-holder (codex finding a): a position opened in pretest still OPEN at
   247	    # test_start whose CENSORED hold already exceeds the swing cap is a week-holder invisible to the
   248	    # median (the close lands after the boundary). Reject on the censored lower-bound alone.
   249	    cmh = jm.get("censored_max_hold_s", 0.0)
   250	    if cmh > SWING_MAX_HOLD_S:
   251	        f.append(f"hold_too_long_censored (>{SWING_MAX_HOLD_S:.0f}s, open-at-test_start {cmh:.0f}s)")
   252	    if jm["share_below_latency"] > SHARE_BELOW_LATENCY_CAP:
   253	        f.append(f"share_below_latency>{SHARE_BELOW_LATENCY_CAP:.0%} ({jm['share_below_latency']:.0%})")
   254	    # LEVERAGE gate (Alberto spec): copy high but not ultra-high; reject median daily leverage > 10x.
   255	    mlev = eqm.get("median_leverage", 0.0)
   256	    if equity_required and mlev > LEVERAGE_CAP:
   257	        f.append(f"leverage>{LEVERAGE_CAP:.0f}x ({mlev:.1f}x)")
   258	    # CONSISTENCY gate (Alberto spec): >=80% of active days green, enforced only when enough active days
   259	    # exist to judge it (short fold windows can't prove consistency; thin history caught by other gates).
   260	    nad = int(eqm.get("n_active_days", 0))
   261	    fdg = eqm.get("frac_days_green", 0.0)
   262	    if equity_required and nad >= MIN_ACTIVE_DAYS_GREEN and fdg < DAYS_GREEN_MIN:
   263	        f.append(f"days_green<{DAYS_GREEN_MIN:.0%} ({fdg:.0%} over {nad}d)")
   264	    # accessibility: unknown (NaN) does NOT fire (loose); only fail when known and < min
   265	    af = jm["accessible_frac_notional"]
   266	    if af == af and af < ACCESSIBLE_FRAC_MIN:
   267	        f.append(f"accessible_frac<{ACCESSIBLE_FRAC_MIN:.0%} ({af:.0%})")
   268	    if not equity_required:
   269	        # Fixed-notional/direction-only lane: M1-derived equity, DD, leverage,
   270	        # and consistency are intentionally absent. Source net PnL is recorded
   271	        # but final performance judgment belongs to after-cost M7/M6b.
   272	        blocking = [x for x in f if not x.startswith("net_pnl<=0")]
   273	        return (len(blocking) == 0, f)
   274	    if COPYABILITY_ONLY:
   275	        # eligible if NO risk/copyability fail; performance fails recorded but non-blocking.
   276	        blocking = [x for x in f if not x.startswith(_PERF_FAIL_PREFIXES)]
   277	        return (len(blocking) == 0, f)
   278	    return (len(f) == 0, f)

   281	def run(folds: pd.DataFrame, journeys: pd.DataFrame, equity: pd.DataFrame, m04: pd.DataFrame | None,
   282	        accessible_by_fold: dict | None = None,
   283	        active_test_folds_by_wallet: dict | None = None,
   284	        m04_by_fold: dict[int, pd.DataFrame] | None = None,
   285	        quarantined_wallets: set[str] | None = None,
   286	        equity_required: bool = True) -> tuple[pd.DataFrame, pd.DataFrame, dict]:
   287	    quarantined_wallets = {w.lower() for w in (quarantined_wallets or set())}
   288	    if len(journeys) and {"lifecycle_valid", "stream_replay_valid"}.issubset(journeys.columns):
   289	        journeys = journeys[
   290	            journeys["lifecycle_valid"].fillna(False).astype(bool)
   291	            & journeys["stream_replay_valid"].fillna(False).astype(bool)
   292	        ].copy()
   293	    # entities to evaluate: M4 copyable primaries — defensive filter on all three (codex code-r1 #2).
   294	    if m04_by_fold is None:
   295	        if m04 is None:
   296	            raise ValueError("run() requires either m04 or m04_by_fold")
   297	        prim = _prim_from_m04(m04)
   298	        prim_by_fold = None
   299	        logger.warning("LOUD WARNING: using a single M4 authenticity file for all folds; "
   300	                       "M5 selection is not fold-pure. Pass --m04-dir for fold-pure M4.")
   301	        logger.info(f"M4 copyable primaries: {len(prim)} entities")
   302	    else:
   303	        prim_by_fold = {int(fid): _prim_from_m04(df) for fid, df in m04_by_fold.items()}
   304	        prim = (pd.concat(prim_by_fold.values(), ignore_index=True).drop_duplicates("entity_id")
   305	                if prim_by_fold else pd.DataFrame())
   306	        logger.info("fold-pure M4 enabled: copyable primaries per fold = %s",
   307	                    {fid: len(df) for fid, df in prim_by_fold.items()})
   308	    jr_by = {w: g for w, g in journeys.groupby("wallet")} if len(journeys) else {}
   309	    # codex code-r1 #6: precompute date_ms ONCE (no per-entity per-fold .apply(_ms)).
   310	    eq_by = {}
   311	    if len(equity):
   312	        eq = equity.copy()
   313	        # Force ms resolution then int64 = epoch ms (consistent with _ms). pandas 2.x infers
   314	        # datetime64[s] from date objects, so a bare astype(int64) yields SECONDS — wrong by 10^6.
   315	        eq["date_ms"] = pd.to_datetime(eq["date"]).astype("datetime64[ms]").astype("int64")
   316	        eq_by = {w: g for w, g in eq.groupby("wallet")}
   317	
   318	    rows = []
   319	    waterfall = {"folds": {}}
   320	    for _, fr in folds.sort_values("fold_id").iterrows():
   321	        fid = int(fr["fold_id"])
   322	        lo_ms, hi_ms = _ms(fr["train_start"]), _ms(fr["test_start"])  # pretest = [train_start, test_start)
   323	        acc = (accessible_by_fold or {}).get(fid)            # set or None(unknown)
   324	        acc_as_of = hi_ms if acc is not None else None
   325	        n_eval = n_elig = 0
   326	        fold_prim = prim_by_fold.get(fid, pd.DataFrame()) if prim_by_fold is not None else prim
   327	        for _, p in fold_prim.iterrows():
   328	            w = p["wallet"]
   329	            if w in eq_by:
   330	                g = eq_by[w]
   331	                eqm = flow_adjusted_twr(g[(g["date_ms"] >= lo_ms) & (g["date_ms"] < hi_ms)])
   332	            else:
   333	                eqm = {"roe": 0.0, "max_dd": 0.0, "n_days": 0, "structural_ruin": True,
   334	                       "median_equity": 0.0, "frac_days_green": 0.0, "n_active_days": 0,
   335	                       "median_leverage": 0.0}
   336	            jm = journey_metrics(jr_by.get(w), lo_ms, hi_ms, acc)
   337	            elig, fails = apply_floors(eqm, jm, equity_required=equity_required)
   338	            recon_quarantined = equity_required and w.lower() in quarantined_wallets
   339	            if recon_quarantined:
   340	                elig = False
   341	                fails.append("reconstruction_quarantined")
   342	            n_eval += 1; n_elig += int(elig)
   343	            rows.append({
   344	                "entity_id": int(p["entity_id"]), "primary_wallet": w, "fold_id": fid,
   345	                "eligible": elig, "fail_reasons": "|".join(fails),
   346	                "net_pnl_pretest": jm["net_pnl"], "roe_pretest_flow_adj": eqm["roe"],
   347	                "median_equity_pretest": eqm.get("median_equity", 0.0),  # codex finding c: auditability
   348	                "max_dd_pretest": eqm["max_dd"], "structural_ruin_flag": eqm["structural_ruin"],
   349	                "median_leverage_pretest": eqm.get("median_leverage", 0.0),
   350	                "n_incomplete_equity_days_excluded": eqm.get("n_incomplete_excluded", 0),
   351	                "reconstruction_quarantined": recon_quarantined,
   352	                "frac_days_green_pretest": eqm.get("frac_days_green", 0.0),
   353	                "n_active_days_pretest": eqm.get("n_active_days", 0),
   354	                "n_journeys_pretest": jm["n_journeys"], "median_hold_s_pretest": jm["median_hold_s"],
   355	                "share_dur_below_latency": jm["share_below_latency"],
   356	                "accessible_frac_notional": jm["accessible_frac_notional"],
   357	                "accessible_frac_count": jm["accessible_frac_count"],
   358	                "accessibility_unknown": acc is None,
   359	                "accessible_set_as_of_ms": acc_as_of,
   360	                "m4_tier": p["tier"], "q_codes": ",".join(c for c in str(p["reason_codes"]).split(",") if c.startswith("q:")),
   361	                "as_of_ms": hi_ms,
   362	                "selection_lane": "equity" if equity_required else "copyability",
   363	            })
   364	        waterfall["folds"][fid] = {"evaluated": n_eval, "eligible": n_elig}
   365	        logger.info(f"  fold {fid}: {n_eval} evaluated -> {n_elig} eligible")
   366	    elig_df = pd.DataFrame(rows)
   367	
   368	    # pool summary (cross-fold DIAGNOSTIC; G5 enforced downstream at M6/M9, not here)
   369	    # codex code-r1 #6: eligible_folds via one groupby, not a per-entity scan.
   370	    elig_folds_by_eid = (elig_df.groupby("entity_id")["eligible"].sum().astype(int).to_dict()
   371	                         if len(elig_df) else {})
   372	    atf = active_test_folds_by_wallet or {}
   373	    pool_rows = []
   374	    for _, p in prim.iterrows():
   375	        w = p["wallet"]; eid = int(p["entity_id"])
   376	        full_jr = jr_by.get(w)
   377	        total_nj = int((full_jr["exit_ts"].notna()).sum()) if full_jr is not None else 0
   378	        full_roe = flow_adjusted_twr(eq_by[w])["roe"] if w in eq_by else 0.0
   379	        active_tf = atf.get(w, np.nan)
   380	        active_tf_known = bool(pd.notna(active_tf))   # codex code-r2: PER-WALLET, not global
   381	        # codex code-r1 #1: G5 candidate REQUIRES active_test_folds>=3 too (from M3).
   382	        recon_quarantined = w.lower() in quarantined_wallets
   383	        g5 = bool(equity_required and not recon_quarantined and active_tf_known and active_tf >= 3
   384	                  and total_nj >= 5 and full_roe >= ROE_FULL_FLOOR_G5)
   385	        pool_rows.append({
   386	            "entity_id": eid, "primary_wallet": w,
   387	            "eligible_folds": elig_folds_by_eid.get(eid, 0),
   388	            "active_test_folds": active_tf, "total_n_journeys": total_nj,
   389	            "source_6m_roe_full": full_roe,
   390	            "g5_pool_candidate_pass": g5,
   391	            "g5_incomplete": (not equity_required) or not active_tf_known,
   392	            "selection_lane": "equity" if equity_required else "copyability",
   393	            "reconstruction_quarantined": recon_quarantined,
   394	        })
   395	    pool_df = pd.DataFrame(pool_rows)
   396	    return elig_df, pool_df, waterfall

   419	        raise ValueError(
   420	            "M2 journeys lack lifecycle/stream replay validity; rebuild M2"
